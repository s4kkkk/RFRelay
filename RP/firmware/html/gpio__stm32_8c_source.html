46static int gpio_stm32_pins_configure(const struct device* port,
47 gpio_pins_t pins,
48 gpio_flags_t flags)
49{
50 struct gpio_stm32_config* config = (struct gpio_stm32_config* ) port->config;
51
52 uint32_t crx_value = 0;
53 uint8_t pull_flag = 0;
54
55 if (flags & GPIO_INPUT) {
56 crx_value = 0b0100;
57 }
58 else if (flags & GPIO_OUTPUT) {
59 crx_value = 0b0011;
60 }
61 else if ( (flags & GPIO_INPUT_PULLDOWN) || (flags & GPIO_INPUT_PULLUP) ) {
62 crx_value = 0b1000;
63 pull_flag = 1;
64 }
65
66 uint32_t crl_mask = 0, crl_value = 0;
67 uint32_t crh_mask = 0, crh_value = 0;
68 uint32_t odr_mask = 0, odr_value = 0;
69 for (uint8_t i = 0; i < 16; i++) {
70 if ( pins & (0b1 << i) ) {
71 /* Пин с номером i присутствует в требуемом наборе */
72 if (i < 8) {
73 /* устаналиванием биты в маске */
74 crl_mask |= ((GPIO_CRL_MODE0_Msk | GPIO_CRL_CNF0_Msk) << (4*i));
75 /* устанавливаем необходимую конфигурацию */
76 crl_value |= crx_value << (4*i);
77 }
78 else {
79 /* устанавливанием биты в маске */
80 crh_mask |= ((GPIO_CRH_MODE8_Msk |
81 GPIO_CRH_CNF8_Msk) << ( 4*(i-8) ) );
82
83 /* устанавливаем необходимую конфигурацию */
84 crh_value |= crx_value << (4*i);
85 }
86
87 if (pull_flag) {
88 odr_mask |= 0b1<<i;
89 if (flags & GPIO_INPUT_PULLUP) {
90 odr_value |= 0b1 << i;
91 }
92 }
93
94 }
95 }
96
97 /* обнуляем немаскированные поля */
98 config->gpio_regs->CRL &= ~(crl_mask);
99 config->gpio_regs->CRH &= ~(crh_mask);
100 /* записываем значения */
101 config->gpio_regs->CRL |= crl_value;
102 config->gpio_regs->CRH |= crh_value;
103
104 if (pull_flag) {
105 /* необходимо записать в ODR-регистр */
106 config->gpio_regs->ODR &= ~(odr_mask);
107 config->gpio_regs->ODR |= odr_value;
108 }
109
110 return 0;
111}

116static int gpio_stm32_pin_get_config(const struct device* port,
117 gpio_pin_t pin,
118 gpio_flags_t* flags)
119{
120 struct gpio_stm32_config* config = (struct gpio_stm32_config* ) port->config;
121
122 uint32_t crx_value = 0;
123 if (pin < 8) {
124 crx_value = config->gpio_regs->CRL;
125 crx_value = crx_value >> pin*4;
126 }
127 else {
128 crx_value = config->gpio_regs->CRH;
129 crx_value = crx_value >> pin*4;
130 }
131
132 if (crx_value == 0b0100) {
133 *flags = GPIO_INPUT;
134 }
135 else if (crx_value == 0b0011) {
136 *flags = GPIO_OUTPUT;
137 }
138 else if (crx_value == 0b1000) {
139 if ( config->gpio_regs->ODR & (0b1<<pin)) {
140 *flags = GPIO_INPUT_PULLUP;
141 }
142 else {
143 *flags = GPIO_INPUT_PULLDOWN;
144 }
145 }
146
147 return 0;
148}
